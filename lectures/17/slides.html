def is_solvable(tiles):
    # Compter les inversions dans la liste à plat des tuiles (en excluant la case vide)
    # Hypothèse : le plateau est carré et sa largeur est impaire.
    inversions = 0
    for i in range(len(tiles)):
        for j in range(i + 1, len(tiles)):
            if tiles[i] != 0 and tiles[j] != 0 and tiles[i] > tiles[j]:
                inversions += 1
    return inversions % 2 == 0

    return tuple(tiles)

def is_empty(frontier):
    """Vérifie si la frontière est vide."""
    return len(frontier) == 0

def is_goal(state, goal_state):
    """Détermine si un état donné correspond à l'état objectif."""
    return state == goal_state

def expand(state):
    """Génère les états successeurs en déplaçant la case vide dans toutes les directions possibles."""
    size = int(len(state) ** 0.5) # Déterminer la taille du puzzle (3 pour le 8-Puzzle, 4 pour le 15-Puzzle)
    idx = state.index(0) # Trouver l'index de la case vide représentée par 0
    x, y = idx % size, idx // size # Convertir l'index en coordonnées (x, y)
    neighbors = []

    # Définir les mouvements possibles : Gauche, Droite, Haut, Bas
    moves = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    for dx, dy in moves:
        nx, ny = x + dx, y + dy
        # Vérifier si la nouvelle position est dans les limites du puzzle
        if 0 <= nx < size and 0 <= ny < size:
            n_idx = ny * size + nx
            new_state = list(state)
            # Échanger la case vide avec la tuile adjacente
            new_state[idx], new_state[n_idx] = new_state[n_idx], new_state[idx]
            neighbors.append(tuple(new_state))
    return neighbors

def print_solution(solution):
    """Affiche la séquence d'étapes de l'état initial à l'état objectif."""
    size = int(len(solution[0]) ** 0.5)
    for step, state in enumerate(solution):
        print(f"Étape {step} :")
        for i in range(size):
            row = state[i*size:(i+1)*size]
            print(' '.join(str(n) if n != 0 else ' ' for n in row))
        print()

def bfs(initial_state, goal_state):
    frontier = deque() # Initialiser la file d'attente pour BFS
    frontier.append((initial_state, [])) # Un nœud : (état, états précédents)

    reached = {initial_state}

    expanded = 0 # Nombre de nœuds non objectifs dont les successeurs sont générés

    while not is_empty(frontier):
        current_state, path = frontier.popleft()

        if is_goal(current_state, goal_state):
            print(f"Nœuds développés : {expanded}")
            return path + [current_state] # Retourner le chemin réussi

        expanded += 1

        for neighbor in expand(current_state):
            if neighbor not in reached:
                reached.add(neighbor)
                frontier.append((neighbor, path + [current_state]))

    return None

def dfs(initial_state, goal_state):
    frontier = [(initial_state, [])] # Un nœud : (état, états précédents)

    reached = {initial_state}

    expanded = 0

    while not is_empty(frontier):
        current_state, path = frontier.pop()

        if is_goal(current_state, goal_state):
            print(f"Nœuds développés : {expanded}")
            return path + [current_state] # Retourner le chemin réussi

        expanded += 1

        for neighbor in expand(current_state):
            if neighbor not in reached:
                reached.add(neighbor)
                frontier.append((neighbor, path + [current_state]))

    return None

Recherche heuristique

Une recherche informée utilise une estimation du coût restant jusqu’à un état objectif pour guider l’exploration.

Comment une telle estimation pourrait-elle aider à choisir le prochain nœud de la frontière ?

L’estimation utilise des connaissances qui s’ajoutent aux actions, au modèle de transition, au test d’objectif et aux coûts d’action du problème. Un état objectif peut être une ville, un puzzle résolu ou une condition sur un état abstrait ; il n’a pas nécessairement une position physique.

L’estimation guide le choix des possibilités à examiner. Elle ne remplace pas le modèle de transition et ne vérifie pas qu’une solution complète existe.

Heuristiques et priorités de recherche

Pour un nœud de recherche n représentant l’état s :

  • g(n) : coût connu de ce chemin de l’état initial à s.
  • h(s) : estimation du coût restant minimal de s à un état objectif.
  • f(n) : priorité utilisée pour sélectionner une entrée de la frontière.
Stratégie par meilleur choix Priorité f(n)
Recherche gloutonne par meilleur choix h(s)
Recherche à coût uniforme g(n)
Recherche A* g(n)+h(s)

Un état décrit une configuration, tandis qu’un nœud de recherche conserve aussi un chemin pour l’atteindre. Deux nœuds peuvent donc avoir le même état et la même estimation heuristique, mais des coûts de chemin différents. La notation h(n) sert souvent de raccourci pour h(s), où s est l’état conservé dans n.

La recherche gloutonne utilise uniquement le coût restant estimé et n’offre aucune garantie générale de coût minimal. A* considère aussi le coût déjà engagé. La recherche à coût uniforme est le cas particulier d’A* où h=0. Voir Berkeley CS 188, Recherche informée.

La recherche par meilleur choix sélectionne une entrée de la frontière dont f est minimal. L’heuristique h et le coût de chemin g doivent utiliser les mêmes unités de coût.

Une heuristique pour la recherche d’itinéraires

Dans un problème de distance routière, la distance à vol d’oiseau jusqu’à la destination estime la distance restante.

La ligne droite n’a pas à correspondre à une route utilisable. A* ajoute cette estimation à la distance déjà parcourue :

f(n)=g(n)+h(s).

Les deux termes sont mesurés en kilomètres dans l’exemple suivant. La distance à vol d’oiseau est une borne inférieure de la distance routière. Elle n’estimerait pas directement le temps de trajet sans une conversion appropriée en unités de temps.

L’exemple de la Roumanie utilise A* : les priorités affichées incluent la distance parcourue et la distance à vol d’oiseau. Sélectionner uniquement la ville ayant la plus petite distance à vol d’oiseau correspondrait plutôt à la recherche gloutonne par meilleur choix.

Le module heapq de Python retire la plus petite entrée. Pour A*, la première composante est g+h.

BFS comme recherche par meilleur choix

BFS sélectionne les nœuds par profondeur non décroissante. On peut la décrire à l’aide de la fonction d’évaluation

f(n)=\operatorname{depth}(n).

La profondeur est connue à partir du chemin ; ce n’est pas une estimation heuristique du coût restant jusqu’à l’objectif. Lorsque les coûts d’action sont égaux, la profondeur et le coût du chemin induisent le même ordre entre les niveaux. Une file FIFO est une implémentation de BFS plus simple qu’un tas et conserve son ordre habituel à l’intérieur d’un niveau.

On peut décrire DFS de manière semblable en donnant la priorité aux nœuds plus profonds, à l’aide de la profondeur négative et d’une règle appropriée de départage « dernier entré, premier sorti ». Le choix de la règle de départage compte lorsque plusieurs entrées ont la même priorité.

Recherche A*

f(n)=g(n)+h(s).

  • g(n) est le coût du chemin particulier conservé dans l’entrée n.
  • h(s) estime le coût restant minimal depuis son état s.
  • A* sélectionne une entrée de la frontière dont f est minimal.

Connaître exactement g(n) pour un chemin enregistré ne signifie pas qu’il s’agit du chemin le moins coûteux vers cet état. Un autre itinéraire peut l’améliorer. Le dictionnaire de l’implémentation suivante conserve le plus petit coût découvert jusqu’à présent.

L’implémentation du 8-Puzzle utilise des coûts d’action unitaires ; prolonger un chemin fait donc passer g à g+1. La même idée s’applique à d’autres modèles de coût en ajoutant le coût réel de l’action.

Implémentation d’A*

def astar(initial_state, goal_state, heuristic):
    # Les entrées contiennent (f, g, état, états précédents).
    frontier = [(heuristic(initial_state, goal_state), 0, initial_state, [])]
    best_g = {initial_state: 0}
    expanded = 0

    while frontier:
        f, g, current_state, path = heapq.heappop(frontier)
        # Un chemin moins coûteux peut avoir remplacé cette entrée de la file.
        if g > best_g[current_state]:
            continue

        if is_goal(current_state, goal_state):
            return path + [current_state], expanded

        expanded += 1
        for neighbor in expand(current_state):
            new_g = g + 1  # Chaque déplacement du puzzle coûte 1.
            if neighbor not in best_g or new_g < best_g[neighbor]:
                best_g[neighbor] = new_g
                new_f = new_g + heuristic(neighbor, goal_state)
                heapq.heappush(frontier, (new_f, new_g, neighbor, path + [current_state]))

    return None, expanded

La fonction retourne (solution_states, expanded). Une solution inclut les états initial et objectif ; son nombre de déplacements est donc len(solution_states) - 1.

Conserver les chemins de moindre coût

  • best_g[state] conserve le plus petit coût de chemin découvert jusqu’à présent, et non un coût définitif.
  • Ajouter un successeur à la file lorsque son nouveau coût de chemin est strictement inférieur.
  • Ignorer une entrée retirée si son g enregistré est supérieur à best_g[state].
  • Un chemin moins coûteux peut provoquer la réouverture d’un état déjà développé.
  • Appliquer le test d’objectif uniquement après avoir retiré une entrée à jour de valeur f minimale.

Une entrée de la frontière représente un chemin particulier vers un état. Plusieurs entrées peuvent désigner le même état, avec des coûts de chemin différents. Lorsqu’un meilleur chemin est trouvé, la nouvelle entrée est ajoutée ; une entrée plus ancienne et plus coûteuse peut rester dans le tas. Le test des entrées obsolètes l’écarte avant le test d’objectif ou le développement. Par exemple, si un état est placé dans la file à un coût de 7, puis atteint à un coût de 5, best_g devient 5 ; l’ancienne entrée de coût 7 est ignorée lorsqu’elle est retirée.

Un état développé n’est pas définitivement fermé dans cette implémentation. Si son coût enregistré s’améliore, ses successeurs doivent être considérés à nouveau avec ce chemin moins coûteux. Le résultat expanded compte les développements réels de nœuds non objectifs, y compris les développements répétés ; il ne compte ni les entrées obsolètes ni l’état objectif sélectionné.

Pour le 8-Puzzle, chaque action coûte un ; le nouveau coût de chemin est donc g + 1. Lorsque les coûts d’action diffèrent, la même règle utilise plutôt g + action_cost.

La mise à jour du meilleur coût suit la règle présentée dans AIMA, Figure 3.7.

Admissibilité

Une heuristique est admissible si elle ne surestime jamais le coût restant minimal d’une solution :

0\le h(s)\le h^\star(s).

Ici, h^\star(s) est le véritable coût minimal de s à un état objectif.

Une surestimation peut faire paraître peu attrayante une solution réellement peu coûteuse ; A* peut alors retourner une solution plus coûteuse. Une surestimation ne force pas une mauvaise réponse dans chaque cas ; elle supprime la garantie générale.

À tout état objectif G, l’admissibilité donne h(G)=0, car le coût restant est nul. Cette présentation utilise des valeurs heuristiques non négatives. Si aucun état objectif n’est accessible depuis un état, son véritable coût restant est infini.

La condition sur l’heuristique ne constitue pas à elle seule tout l’algorithme : l’implémentation doit aussi conserver les chemins améliorés et appliquer le test d’objectif au bon moment.

Que pourrait-il se passer si l’estimation surestimait le coût restant ?

Optimalité en coût

Un algorithme est optimal en coût lorsqu’il retourne une solution de coût de chemin minimal chaque fois qu’il retourne une solution.

Pour la recherche A* sur graphe ci-dessus, la garantie combine :

  • une heuristique admissible ;
  • les mises à jour du meilleur coût et la réouverture ;
  • un test d’objectif sur une entrée à jour de valeur f minimale.

La preuve concerne la même implémentation que celle utilisée dans les exemples et les expériences. Le test des entrées obsolètes garantit qu’une entrée dépassée n’est pas traitée comme le meilleur itinéraire actuel vers son état. La complétude pose en plus la question de savoir si l’algorithme trouve une solution lorsqu’il en existe une ; les espaces finis et infinis exigent des hypothèses de terminaison différentes.

Théorème : optimalité en coût

Considérons le graphe fini des états accessibles de l’implémentation ci-dessus, avec des coûts d’action unitaires. Soit h^\star(s) le coût restant minimal de l’état s à un état objectif.

L’heuristique est admissible :

0 \le h(s) \le h^\star(s), \qquad h(G)=0 \text{ pour tout état objectif }G.

Avec les règles de meilleur coût et de réouverture ci-dessus, A* retourne une solution de coût minimal lorsqu’il en existe une et signale un échec sinon.

Le 8-Puzzle, fini et à coûts unitaires, satisfait ces hypothèses. Le résultat s’étend aussi aux graphes finis dont les coûts d’action sont non négatifs, lorsque l’implémentation ajoute le coût réel de chaque action. Si aucun état objectif n’est accessible depuis un état, son coût restant jusqu’à un état objectif est infini ; l’admissibilité n’y impose aucune borne supérieure finie. L’heuristique utilisée par l’implémentation est une estimation fixe et non négative pour chaque état.

Les règles de gestion des chemins comptent autant que l’admissibilité. Écarter tout chemin ultérieur vers un état simplement parce qu’il a été généré plus tôt ne satisfait pas ce théorème.

Preuve

1. Supposons qu’A* sélectionne un état objectif sous-optimal.

Soit C^\star le coût de la solution optimale. Supposons que l’algorithme retourne une entrée objectif G de coût C>C^\star.

Puisque h(G)=0,

f(G)=g(G)+h(G)=C.

Examinons la frontière immédiatement avant la sélection de cette entrée objectif à jour.

Preuve

2. Une continuation de coût optimal reste dans la frontière.

Suivons un chemin optimal depuis le départ. Considérons son premier état qui n’a pas encore été développé au coût optimal de son préfixe.

Une entrée à jour de la frontière atteint cet état avec ce coût optimal de préfixe.

L’état initial entre dans la frontière avec un coût nul. Chaque fois qu’un état du chemin optimal choisi est développé au coût optimal de son préfixe, la génération de l’action suivante propose à l’état suivant son propre coût optimal de préfixe.

Si cela améliore best_g, la nouvelle entrée est ajoutée à la file, même si l’état a été développé plus tôt par un chemin plus coûteux. Si le même coût optimal était déjà enregistré, son entrée attend encore ou a déjà été développée à ce coût. Dans le second cas, continuons le long du chemin. Aucun préfixe moins coûteux ne peut exister, car remplacer ce préfixe produirait une solution moins coûteuse.

Le premier état qui n’a pas encore été développé au coût optimal de son préfixe a donc une entrée en attente à ce coût. Elle n’est pas obsolète : best_g ne peut pas devenir inférieur au coût optimal du préfixe. Un tel état doit exister, à moins qu’un état objectif optimal ait déjà été sélectionné ; dans ce cas, l’algorithme aurait déjà retourné une solution.

Un développement antérieur à un coût plus élevé n’invalide pas cet argument : la règle d’amélioration stricte provoque la réouverture de l’état lorsque son préfixe optimal est découvert.

Preuve

3. La borne inférieure de cette entrée de la frontière ne dépasse pas C^\star.

Notons n cette entrée et s son état. Son g(n) enregistré est le coût optimal du préfixe. Par admissibilité,

f(n)=g(n)+h(s) \le g(n)+h^\star(s) =C^\star.

Le préfixe et la continuation restante de coût minimal forment ensemble une solution optimale.

Preuve

4. La règle du f minimal contredit la sélection de l’état objectif sous-optimal.

La frontière contient une entrée à jour n telle que

f(n) \le C^\star < C = f(G).

A* doit sélectionner une entrée dont f est inférieur avant G. Il ne peut donc pas retourner un état objectif sous-optimal.

Sur un graphe fini, les cycles de coût non négatif ne peuvent pas améliorer le coût d’un chemin. Accepter uniquement les améliorations strictes empêche d’ajouter indéfiniment des cycles de coût égal ; seul un nombre fini de chemins améliorés peut être accepté. La recherche se termine donc. Si elle épuise la frontière, il ne reste aucune solution à trouver.

Sur un graphe infini, l’admissibilité seule ne garantit pas la complétude. Un ensemble classique d’hypothèses suffisantes est un branchement fini, l’existence d’une solution de coût fini et des coûts d’action bornés inférieurement par une valeur \varepsilon>0. Puisque h\ge0, seul un nombre fini de chemins candidats peut avoir f\le C^\star sous ces hypothèses. Cela exclut une succession infinie d’étapes de coût nul qui retarderait indéfiniment une solution. Une recherche dans un espace infini sans solution ne termine pas nécessairement en signalant un échec.

Cohérence (consistency)

Une heuristique est cohérente si chaque action de s à s' satisfait

h(s) \le c(s,s')+h(s'), \qquad h(G)=0.

Son coût restant estimé ne peut pas diminuer de plus que le coût de cette action. Par conséquent, le long d’un chemin,

f(n')=g(n)+c(s,s')+h(s') \ge g(n)+h(s)=f(n).

La cohérence est une condition locale sur chaque transition. L’appliquer successivement le long d’un chemin vers un état objectif montre que l’heuristique est admissible. Dans le 8-Puzzle, un déplacement de coût unitaire peut réduire le nombre de tuiles mal placées d’au plus un et peut réduire la distance de Manhattan d’au plus un ; les deux heuristiques sont cohérentes.

Avec une heuristique cohérente, la première sélection d’une entrée à jour de valeur f minimale pour un état utilise déjà un chemin de coût minimal vers cet état. La réouverture après le développement est donc inutile. La même implémentation d’astar fonctionne toujours ; elle n’a simplement pas besoin de développer à nouveau un état à un meilleur coût.

La cohérence ne justifie pas de considérer comme définitif le coût d’un état dès sa première génération : un chemin moins coûteux peut encore être trouvé avant son premier développement.

Une heuristique admissible mais incohérente peut nécessiter une réouverture. La règle du meilleur coût préserve l’optimalité dans ce cas, même si les développements répétés peuvent augmenter le travail. Une variante de recherche sur graphe qui ferme définitivement les états après leur développement s’appuie plutôt sur la cohérence pour cette garantie. Voir Berkeley CS 188, Recherche informée.

Construire une heuristique pour le 8-Puzzle

Quelles informations sur un plateau permettraient d’estimer le nombre de déplacements restants ?

Chaque action coûte une unité ; l’heuristique doit donc estimer un nombre de déplacements. Les exemples excluent la case vide du décompte des tuiles mal placées et de la somme des distances des tuiles numérotées. Leur objectif est le plateau ordonné standard, avec la case vide dans le coin inférieur droit.

Le plateau présenté contient deux inversions : la tuile 6 vient après les tuiles 7 et 8. Déplacer 6 dans la case vide résout le puzzle en un déplacement. Sur un plateau 3×3, un déplacement permis fait varier le nombre d’inversions de 0 ou de ±2. Un déplacement vertical peut faire passer une tuile numérotée devant deux autres tuiles numérotées dans l’ordre de lecture par rangées. Le contre-exemple montre pourquoi le nombre brut d’inversions n’est pas une borne inférieure du nombre de déplacements ; la variation possible de ±2 ne suffit pas, à elle seule, à prouver la non-admissibilité.

La parité remplit toutefois une autre fonction. Le plateau (1,2,3,4,5,6,0,8,7) a une parité d’inversion impaire et ne peut pas atteindre l’état objectif de parité paire. La résolubilité et le coût restant d’une solution sont deux questions différentes.

Tuiles mal placées

def misplaced_tiles_distance(state, goal_state):

    # Compter le nombre de tuiles mal placées

    misplaced_tiles = sum(1 for s, g in zip(state, goal_state) if s != g and s != 0)
    
    return misplaced_tiles

Un déplacement permis change la position d’exactement une tuile numérotée. Il peut corriger au plus une tuile mal placée ; les autres tuiles numérotées conservent leur position. Si h(s) tuiles sont mal placées, toute solution exige donc au moins h(s) déplacements, d’où h(s)\le h^\star(s).

La même observation établit la cohérence : un déplacement peut réduire le nombre de tuiles mal placées d’au plus une unité, donc h(s)\le1+h(s'). La valeur de l’état objectif est nulle.

Cette heuristique est-elle admissible ? La case vide est exclue.

Distance de Manhattan

def manhattan_distance(state, goal_state):
    distance = 0
    size = int(len(state) ** 0.5)
    for num in range(1, len(state)):
        idx1 = state.index(num)
        idx2 = goal_state.index(num)
        x1, y1 = idx1 % size, idx1 // size
        x2, y2 = idx2 % size, idx2 // size
        distance += abs(x1 - x2) + abs(y1 - y2)
    return distance

h_{\mathrm{Manhattan}}(s)= \sum_{t=1}^{8}\left(|x_t-x_t^\star|+|y_t-y_t^\star|\right).

Pour la tuile t, (x_t,y_t) est sa position actuelle et (x_t^\star,y_t^\star) est sa position dans l’état objectif. La somme porte sur les huit tuiles numérotées.

Une action permise déplace une tuile numérotée d’une case, faisant varier sa contribution à la distance de Manhattan d’exactement une unité, à la hausse ou à la baisse. La somme peut diminuer d’au plus une unité par déplacement. Pour atteindre une somme nulle, il faut donc au moins autant de déplacements que la somme initiale des distances de Manhattan, ce qui prouve l’admissibilité. La même observation établit la cohérence : h(s)\le1+h(s').

L’interprétation par un problème relâché rend les tuiles indépendantes : chaque tuile numérotée peut se déplacer horizontalement ou verticalement sans avoir besoin de la case vide, et les contraintes d’occupation sont ignorées. Dans ce problème relâché, chaque tuile peut suivre un plus court chemin sur la grille jusqu’à sa cible ; le coût total exact est donc la somme des distances de Manhattan. Retirer des contraintes ne peut pas augmenter le coût minimal d’une solution, ce qui fournit une borne inférieure pour le puzzle réel.

La case vide est exclue. Pourquoi cette somme est-elle une borne inférieure du nombre de déplacements nécessaires ?

Comparer les heuristiques

Pour tout état atteignable, en excluant la case vide,

h_{\mathrm{misplaced}}(s) \le h_{\mathrm{Manhattan}}(s) \le h^\star(s).

  • Chaque tuile numérotée mal placée contribue au moins une unité à la distance de Manhattan.
  • Manhattan domine donc le nombre de tuiles mal placées : c’est une borne inférieure plus informative.
  • Cela garantit-il un temps d’exécution réel plus court pour chaque instance ?

Une estimation admissible plus élevée peut écarter davantage de chemins peu prometteurs. Elle ne produit pas nécessairement strictement moins de développements pour chaque problème. Les entrées dont la priorité égale le coût optimal peuvent être traitées différemment selon les règles de départage, et le calcul d’une heuristique plus informative peut coûter plus cher par nœud.

Les expériences gardent fixes l’algorithme, les instances du problème et la règle de départage, et mesurent les développements de nœuds non objectifs. Elles ne mesurent pas le temps d’exécution écoulé. Voir Berkeley CS 188, Dominance.

La frontière peut contenir des entrées correspondant à un état objectif. Leur valeur heuristique est nulle ; leur priorité est donc le coût du chemin complet qu’elles représentent. Exploration : valider l’optimalité

BFS retourne une solution avec un nombre minimal de déplacements lorsque chaque action coûte un.

  1. Générez un échantillon de taille limitée de plateaux résolubles du 8-Puzzle.
  2. Trouvez une solution avec BFS et une solution avec A* pour chaque plateau.
  3. Vérifiez que chaque chemin est légal, atteint l’état objectif et comporte le même nombre de déplacements.
  4. Répétez avec les deux heuristiques admissibles.

BFS fournit une référence indépendante pour le coût de la solution; il n’est pas nécessaire qu’il suive le même chemin que A*. Plusieurs solutions optimales peuvent exister. Limitez la taille de l’échantillon de validation, car BFS peut visiter une grande partie de l’espace d’états accessible pour chaque plateau.

Exploration : une heuristique nulle

Avec h(s)=0, A* ordonne sa frontière uniquement selon g(n) : il devient une recherche à coût uniforme.

Lorsque toutes les actions coûtent un, la recherche à coût uniforme et BFS développent les nœuds par profondeur non décroissante.

Les deux méthodes donnent le même coût minimal de solution, mais leur ordre au sein d’une même profondeur peut différer. La règle de départage du tas, l’ordre des successeurs, la politique de gestion des doublons et l’emplacement du test d’état objectif influencent l’ordre et le travail effectué avant la sélection de l’état objectif. La recherche dans un graphe évite les états répétés; elle n’énumère pas tous les chemins possibles.

Mesurer les performances

  • Complétude : L’algorithme trouve-t-il une solution dès lors qu’il en existe une?
  • Optimalité en coût : Retourne-t-il une solution dont le coût du chemin est minimal?

Sur un graphe fini d’états accessibles, les implémentations de BFS et DFS avec détection des états répétés sont complètes. L’implémentation corrigée de A* est également complète sous les hypothèses du théorème. Elles peuvent épuiser la frontière et signaler un échec lorsqu’aucun état objectif n’est accessible.

Dans les espaces infinis, trouver une solution existante et terminer pour tout problème sans solution sont deux exigences différentes. DFS peut suivre une branche infinie. Les hypothèses usuelles de complétude de A* comprennent un facteur de branchement fini, une solution de coût fini et des coûts d’action minorés par une constante strictement positive.

BFS garantit un coût optimal lorsque les coûts d’action sont égaux. DFS n’offre aucune garantie générale d’optimalité en coût. A* repose sur l’admissibilité et sur les politiques précisées de conservation des meilleurs coûts, de réouverture et de sélection d’un état objectif.

Complexité en temps et en espace

  • Graphe de l’espace d’états : utilisez le nombre d’états V et d’actions E.
  • Arbre de recherche : utilisez le facteur de branchement b et la profondeur de solution d.
  • Implémentation réelle : tenez aussi compte des tas, de l’évaluation des heuristiques et des chemins mémorisés.

Avec des opérations sur les états en temps constant et des enregistrements de taille constante par nœud, la recherche BFS ordinaire dans un graphe prend un temps \mathcal{O}(V+E) et un espace \mathcal{O}(V). Le code de la présentation copie des chemins complets dans les entrées de la frontière, ce qui ajoute du travail et de la mémoire proportionnels à leurs longueurs; ces coûts doivent être pris en compte dans la description de l’implémentation.

Pour une recherche BFS sous forme d’arbre avec test d’état objectif à la génération, la borne habituelle est de \mathcal{O}(b^d) nœuds générés. Avec un test au retrait de la frontière, comme dans notre code, des nœuds à la profondeur d+1 peuvent être générés avant qu’un état objectif à la profondeur d ne soit sélectionné. L’analyse correspondante de l’arbre dans le pire cas donne alors \mathcal{O}(b^{d+1}) occurrences générées ou mémorisées. Les bornes sur le graphe de l’espace d’états et celles sur l’arbre de recherche comptent des objets différents.

A* ajoute des opérations de file de priorité et des évaluations de l’heuristique. Avec des heuristiques admissibles mais incohérentes, il peut aussi développer à nouveau un état après la découverte d’un chemin amélioré. Le nombre de développements peut donc dépasser le nombre d’états distincts développés. Une petite frontière n’élimine pas la mémoire conservée dans les enregistrements des meilleurs coûts et dans les chemins copiés.

Les exemples indiquent les développements de nœuds. Une comparaison complète des performances doit aussi tenir compte du travail et de la mémoire utilisés par chaque développement.

Épilogue

Résumé

  • Les priorités de la recherche par meilleur choix distinguent la recherche gloutonne, la recherche à coût uniforme et A*.
  • A* combine le coût connu du chemin et une estimation du coût restant.
  • Les mises à jour des meilleurs coûts, la réouverture et la sélection d’un état objectif relient l’implémentation à la preuve d’optimalité.
  • L’admissibilité, la cohérence et la dominance permettent de justifier et de comparer les heuristiques du 8-Puzzle.

Une entrée de la frontière de A* représente un chemin particulier. best_g enregistre le plus faible coût découvert vers chaque état, tandis que des entrées obsolètes peuvent rester dans le tas jusqu’à leur retrait. Un chemin amélioré peut rouvrir un état. Ces politiques préservent l’entrée de la frontière sur un chemin optimal utilisée dans la preuve.

Le nombre de tuiles mal placées et la distance de Manhattan excluent la case vide. Les deux heuristiques sont admissibles et cohérentes pour les déplacements de coût un du 8-Puzzle. Manhattan domine le nombre de tuiles mal placées, mais le coût de calcul et les choix de départage comptent toujours dans la mesure des performances.

Sur le plateau fixe de la présentation 16, BFS et A* avec Manhattan retournent tous deux 25 déplacements, avec respectivement 145 605 et 2 255 nœuds non objectifs développés. DFS retourne un chemin plus long après moins de développements. L’expérience appariée compare les deux heuristiques sur des plateaux identiques et conserve séparément les coûts de leurs solutions.
